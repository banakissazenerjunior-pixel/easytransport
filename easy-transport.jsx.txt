const TARIF_BASE_KM = 45; // FCFA / km approx.

function distanceKm(a, b) {
  const R = 6371;
  const dLat = ((b.lat - a.lat) * Math.PI) / 180;
  const dLng = ((b.lng - a.lng) * Math.PI) / 180;
  const s =
    Math.sin(dLat / 2) ** 2 +
    Math.cos((a.lat * Math.PI) / 180) *
      Math.cos((b.lat * Math.PI) / 180) *
      Math.sin(dLng / 2) ** 2;
  return R * 2 * Math.atan2(Math.sqrt(s), Math.sqrt(1 - s));
}

function villeLaPlusProche(lat, lng) {
  let meilleure = VILLES_GABON[0];
  let meilleureDist = Infinity;
  for (const v of VILLES_GABON) {
    const d = distanceKm({ lat, lng }, v);
    if (d < meilleureDist) {
      meilleureDist = d;
      meilleure = v;
    }
  }
  return meilleure;
}

function genererCodeReservation() {
  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
  let code = "";
  for (let i = 0; i < 6; i++) code += chars[Math.floor(Math.random() * chars.length)];
  return `ET-${code}`;
}

function prixEstime(villeA, villeB, passagers) {
  const va = VILLES_GABON.find((v) => v.nom === villeA);
  const vb = VILLES_GABON.find((v) => v.nom === villeB);
  if (!va || !vb) return 5000 * passagers;
  const km = Math.max(distanceKm(va, vb), 15);
  const prix = Math.round((km * TARIF_BASE_KM) / 500) * 500;
  return Math.max(prix, 2500) * passagers;
}

function formatFCFA(n) {
  return n.toLocaleString("fr-FR").replace(/,/g, " ") + " FCFA";
}

const COMPAGNIES = [
  { nom: "SOVOG Transport", couleur: "#0E3B2E" },
  { nom: "Gabon Voyages", couleur: "#B54834" },
  { nom: "TransEquateur", couleur: "#1C5C43" },
];

function ajouterJours(dateStr, n) {
  const d = new Date(dateStr + "T12:00:00");
  d.setDate(d.getDate() + n);
  return d.toISOString().split("T")[0];
}

function formatDateCourt(dateStr) {
  const d = new Date(dateStr + "T12:00:00");
  return d.toLocaleDateString("fr-FR", { weekday: "short", day: "2-digit", month: "short" });
}

// Génère une liste de départs disponibles (simulée) pour un trajet et une date donnés.
// Déterministe (basé sur la date + les villes) pour que les résultats restent stables.
function genererDeparts(depart, destination, dateStr, passagers) {
  const graine = (depart + destination + dateStr).split("").reduce((s, c) => s + c.charCodeAt(0), 0);
  const prixBase = prixEstime(depart, destination, 1);
  const horaires = ["05:30", "07:00", "09:15", "12:00", "14:30", "17:00"];
  const nbDeparts = 2 + (graine % 3); // 2 à 4 départs ce jour-là

  const departs = [];
  for (let i = 0; i < nbDeparts; i++) {
    const idxCompagnie = (graine + i * 7) % COMPAGNIES.length;
    const idxHoraire = (graine + i * 3) % horaires.length;
    const placesRestantes = ((graine + i * 11) % 18) + 2; // entre 2 et 19
    const variation = 1 + (((graine + i) % 5) - 2) * 0.04; // légère variation de prix
    departs.push({
      id: `${dateStr}-${i}`,
      compagnie: COMPAGNIES[idxCompagnie].nom,
      couleur: COMPAGNIES[idxCompagnie].couleur,
      heure: horaires[idxHoraire],
      placesRestantes,
      prix: Math.round((prixBase * variation) / 500) * 500,
      complet: placesRestantes < passagers,
    });
  }
  return departs.sort((a, b) => a.heure.localeCompare(b.heure));
}

function FormulaireReservation({ onRetour, onSuivant }) {
  const [geoStatut, setGeoStatut] = useState("idle"); // idle | recherche | ok | refuse
  const [depart, setDepart] = useState(null);

  const [nom, setNom] = useState("");
  const [telephone, setTelephone] = useState("");
  const [destination, setDestination] = useState("");
  const [date, setDate] = useState("");
  const [passagers, setPassagers] = useState(1);

  function detecterPosition() {
    setGeoStatut("recherche");
    if (!navigator.geolocation) {
      setGeoStatut("refuse");
      return;
    }
    navigator.geolocation.getCurrentPosition(
      (pos) => {
        const ville = villeLaPlusProche(pos.coords.latitude, pos.coords.longitude);
        setDepart(ville.nom);
        setGeoStatut("ok");
      },
      () => setGeoStatut("refuse"),
      { timeout: 8000 }
    );
  }

  const villesArrivee = VILLES_GABON.filter((v) => v.nom !== depart);
  const pretAValider = depart && nom.trim() && telephone.trim() && destination && date && passagers > 0;

  return (
    <div className="min-h-[100dvh] bg-mist">
      <EntetePage titre="Réserver un billet" sousTitre="Étape 1 sur 3 — Vos informations" onRetour={onRetour} />

      <div className="px-5 pb-32 max-w-md mx-auto space-y-5">
        {/* Bloc géolocalisation */}
        <div className="bg-forest rounded-2xl p-4 text-paper">
          {geoStatut === "idle" && (
            <button
              onClick={detecterPosition}
              className="w-full flex items-center justify-between gap-3"
            >
              <span className="flex items-center gap-3 text-left">
                <span className="h-10 w-10 rounded-full bg-paper/10 flex items-center justify-center shrink-0">
                  <Navigation size={17} />
                </span>
                <span>
                  <span className="block font-semibold text-[14.5px]">Détecter mon départ</span>
                  <span className="block text-[12.5px] text-paper/55">
                    Via la position de votre téléphone
                  </span>
                </span>
              </span>
              <ArrowRight size={17} className="text-gold shrink-0" />
            </button>
          )}
          {geoStatut === "recherche" && (
            <div className="flex items-center gap-3">
              <span className="h-10 w-10 rounded-full bg-paper/10 flex items-center justify-center shrink-0">
                <Navigation size={17} className="animate-pulse" />
              </span>
              <span className="text-[14px] text-paper/75">Localisation en cours…</span>
            </div>
          )}
          {geoStatut === "ok" && (
            <div className="flex items-center gap-3">
              <span className="h-10 w-10 rounded-full bg-gold/20 flex items-center justify-center shrink-0">
                <CheckCircle2 size={18} className="text-gold" />
              </span>
              <span>
                <span className="block text-[12px] text-paper/50 uppercase tracking-wider font-semibold">
                  Ville de départ détectée
                </span>
                <span className="block font-display font-bold text-[16px]">{depart}</span>
              </span>
            </div>
          )}
          {geoStatut === "refuse" && (
            <div>
              <p className="text-[13.5px] text-paper/70 mb-2.5">
                Localisation indisponible. Choisissez votre ville de départ manuellement.
              </p>
              <select
                value={depart || ""}
                onChange={(e) => {
                  setDepart(e.target.value);
                  setGeoStatut("ok");
                }}
                className="w-full rounded-lg bg-paper/10 border-2 border-paper/20 px-3 py-2.5 text-[14px] text-paper outline-none focus:border-gold"
              >
                <option value="" className="text-ink">
                  Sélectionner une ville…
                </option>
                {VILLES_GABON.map((v) => (
                  <option key={v.nom} value={v.nom} className="text-ink">
                    {v.nom}
                  </option>
                ))}
              </select>
            </div>
          )}
        </div>

        <Champ icone={User} label="Nom et prénom">
          <input
            className={inputCls}
            placeholder="Ex. Banakissa Ndong"
            value={nom}
            onChange={(e) => setNom(e.target.value)}
          />
        </Champ>

        <Champ icone={Phone} label="Numéro de téléphone">
          <input
            className={inputCls}
            placeholder="074 XX XX XX"
            inputMode="tel"
            value={telephone}
            onChange={(e) => setTelephone(e.target.value)}
          />
        </Champ>

        <Champ icone={MapPin} label="Ville d'arrivée">
          <select
            className={inputCls + " appearance-none"}
            value={destination}
            onChange={(e) => setDestination(e.target.value)}
            disabled={!depart}
          >
            <option value="">{depart ? "Choisir la destination…" : "Détectez d'abord votre départ"}</option>
            {villesArrivee.map((v) => (
              <option key={v.nom} value={v.nom}>
                {v.nom}
              </option>
            ))}
          </select>
        </Champ>

        <div className="grid grid-cols-2 gap-3">
          <Champ icone={Calendar} label="Date du voyage">
            <input
              type="date"
              className={inputCls}
              min={new Date().toISOString().split("T")[0]}
              value={date}
              onChange={(e) => setDate(e.target.value)}
            />
          </Champ>
          <Champ icone={Users} label="Passagers">
            <div className="flex items-center rounded-xl border-2 border-forest/15 bg-white px-2 py-1.5">
              <button
                onClick={() => setPassagers((p) => Math.max(1, p - 1))}
                className="h-8 w-8 rounded-lg text-forest font-bold text-lg active:bg-forest/10"
              >
                −
              </button>
              <span className="flex-1 text-center font-semibold text-[15px]">{passagers}</span>
              <button
                onClick={() => setPassagers((p) => Math.min(9, p + 1))}
                className="h-8 w-8 rounded-lg text-forest font-bold text-lg active:bg-forest/10"
              >
                +
              </button>
            </div>
          </Champ>
        </div>

        {depart && destination && (
          <div className="bg-white rounded-2xl p-4 border-2 border-forest/10">
            <LigneTrajet depart={depart} arrivee={destination} detectee={geoStatut === "ok"} />
          </div>
        )}
      </div>

      <div className="fixed bottom-0 left-0 right-0 bg-mist/95 backdrop-blur border-t border-forest/10 px-5 py-4">
        <div className="max-w-md mx-auto">
          <button
            disabled={!pretAValider}
            onClick={() =>
              onSuivant({ nom, telephone, depart, destination, date, passagers })
            }
            className="w-full bg-forest disabled:bg-forest/25 text-paper disabled:text-paper/50 font-display font-bold rounded-xl py-4 flex items-center justify-center gap-2 transition active:scale-[0.98]"
          >
            Continuer vers le paiement
            <ArrowRight size={17} />
          </button>
        </div>
      </div>
    </div>
  );
}

function EcranResultats({ reservation, onRetour, onChoisir }) {
  const [chargement, setChargement] = useState(true);
  const [dateActive, setDateActive] = useState(reservation.date);

  const dates = useMemo(
    () => [
      ajouterJours(reservation.date, -1),
      reservation.date,
      ajouterJours(reservation.date, 1),
    ],
    [reservation.date]
  );

  React.useEffect(() => {
    setChargement(true);
    const t = setTimeout(() => setChargement(false), 900);
    return () => clearTimeout(t);
  }, []);

  const departs = useMemo(
    () => genererDeparts(reservation.depart, reservation.destination, dateActive, reservation.passagers),
    [reservation.depart, reservation.destination, dateActive, reservation.passagers]
  );

  const dispoCeJour = departs.some((d) => !d.complet);

  return (
    <div className="min-h-[100dvh] bg-mist">
      <EntetePage
        titre="Disponibilités"
        sousTitre={`${reservation.depart} → ${reservation.destination}`}
        onRetour={onRetour}
      />

      <div className="px-5 max-w-md mx-auto">
        {/* Sélecteur de date */}
        <div className="flex gap-2 mb-5">
          {dates.map((d, i) => (
            <button
              key={d}
              onClick={() => setDateActive(d)}
              className={`flex-1 rounded-xl px-2 py-2.5 text-center transition border-2 ${
                dateActive === d
                  ? "bg-forest border-forest text-paper"
                  : "bg-white border-forest/12 text-ink/70"
              }`}
            >
              <span className="block text-[10.5px] uppercase tracking-wide opacity-70">
                {i === 0 ? "Veille" : i === 1 ? "Souhaitée" : "Lendemain"}
              </span>
              <span className="block text-[13px] font-bold capitalize mt-0.5">
                {formatDateCourt(d)}
              </span>
            </button>
          ))}
        </div>

        {chargement ? (
          <div className="flex flex-col items-center justify-center py-16 text-forest/50">
            <SearchCheck size={28} className="animate-pulse mb-3" />
            <p className="text-[14px]">Recherche des véhicules disponibles…</p>
          </div>
        ) : (
          <div className="space-y-3 pb-10">
            {!dispoCeJour && (
              <div className="bg-rust/10 border-2 border-rust/20 rounded-xl p-4 mb-1">
                <p className="text-[13.5px] text-rust font-semibold">
                  Aucun véhicule avec assez de places ce jour-là.
                </p>
                <p className="text-[12.5px] text-ink/50 mt-0.5">
                  Essayez la veille ou le lendemain via les onglets ci-dessus.
                </p>
              </div>
            )}

            {departs.map((d) => (
              <div
                key={d.id}
                className={`bg-white rounded-2xl p-4 border-2 ${
                  d.complet ? "border-forest/8 opacity-50" : "border-forest/10"
                }`}
              >
                <div className="flex items-center justify-between mb-3">
                  <span className="flex items-center gap-2">
                    <span
                      className="h-8 w-8 rounded-lg flex items-center justify-center text-white text-[11px] font-bold shrink-0"
                      style={{ backgroundColor: d.couleur }}
                    >
                      {d.compagnie.slice(0, 2).toUpperCase()}
                    </span>
                    <span className="font-semibold text-[14px] text-ink">{d.compagnie}</span>
                  </span>
                  <span className="flex items-center gap-1 text-[13px] font-bold text-ink/70">
                    <Clock size={13} />
                    {d.heure}
                  </span>
                </div>

                <div className="flex items-center justify-between">
                  <span className="flex items-center gap-1.5 text-[12.5px] text-ink/50">
                    <Armchair size={14} />
                    {d.complet
                      ? "Places insuffisantes"
                      : `${d.placesRestantes} place(s) restante(s)`}
                  </span>
                  <span className="font-display font-bold text-[15px] text-forest">
                    {formatFCFA(d.prix * reservation.passagers)}
                  </span>
                </div>

                <button
                  disabled={d.complet}
                  onClick={() => onChoisir({ ...reservation, date: dateActive, trajet: d })}
                  className="w-full mt-3 bg-forest disabled:bg-forest/15 disabled:text-ink/30 text-paper text-[13.5px] font-semibold rounded-lg py-2.5 transition active:scale-[0.98]"
                >
                  {d.complet ? "Complet" : "Choisir ce départ"}
                </button>
              </div>
            ))}
          </div>
        )}
      </div>
    </div>
  );
}

                  {op.initiale}
                </span>
                <span className="flex-1 text-left font-semibold text-[15px] text-ink">{op.nom}</span>

                    operateur === op.id ? "border-forest bg-forest" : "border-forest/20"
                  }`}

              </button>
            ))}

        </div>

          <Champ icone={Smartphone} label={`Numéro ${operateur === "airtel" ? "Airtel Money" : "Moov Money"}`}>
            <input className={inputCls} placeholder="Numéro à débiter" inputMode="tel" defaultValue={reservation.telephone} />
          </Champ>
        )}

        <button
          disabled={!operateur || traitement}
          onClick={payer}
          className="w-full bg-gold disabled:bg-gold/40 text-forest font-display font-bold rounded-xl py-4 flex items-center justify-center gap-2 transition active:scale-[0.98]"
        >
          {traitement ? "Confirmation en cours…" : `Payer ${formatFCFA(montant)}`}
        </button>
        <p className="text-center text-[11.5px] text-ink/40">
          Une demande de confirmation sera envoyée sur votre téléphone.
        </p>
      </div>
    </div>
  );
}

/* ------------------------------------------------------------------ */

function ConnexionAdmin({ onRetour, onConnecte }) {
  const [utilisateur, setUtilisateur] = useState("");
  const [motDePasse, setMotDePasse] = useState("");
  const [erreur, setErreur] = useState(false);

  function connecter() {
    if (utilisateur.trim() && motDePasse.trim()) {
      setErreur(false);
      onConnecte();
    } else {
      setErreur(true);
    }
  }

  return (
    <div className="min-h-[100dvh] bg-forest flex flex-col">
      <div className="px-5 pt-6">
        <button
          onClick={onRetour}
          className="h-9 w-9 rounded-full bg-paper/10 flex items-center justify-center active:scale-90 transition"
        >
          <ArrowLeft size={16} className="text-paper" strokeWidth={2.5} />
        </button>
      </div>

      <div className="flex-1 flex flex-col justify-center px-6 max-w-md mx-auto w-full">
        <div className="h-12 w-12 rounded-2xl bg-gold flex items-center justify-center mb-5">
          <Lock size={22} className="text-forest" strokeWidth={2.5} />
        </div>
        <h1 className="font-display font-bold text-[24px] text-paper mb-1.5">Espace transporteur</h1>
        <p className="text-paper/55 text-[14px] mb-8">Connectez-vous pour gérer vos trajets et départs.</p>

        <div className="space-y-3.5">
          <input
            className="w-full rounded-xl bg-paper/8 border-2 border-paper/15 px-4 py-3.5 text-[15px] text-paper placeholder:text-paper/35 outline-none focus:border-gold transition"
            placeholder="Identifiant"
            value={utilisateur}
            onChange={(e) => setUtilisateur(e.target.value)}
          />
          <input
            type="password"
            className="w-full rounded-xl bg-paper/8 border-2 border-paper/15 px-4 py-3.5 text-[15px] text-paper placeholder:text-paper/35 outline-none focus:border-gold transition"
            placeholder="Mot de passe"
            value={motDePasse}
            onChange={(e) => setMotDePasse(e.target.value)}
            onKeyDown={(e) => e.key === "Enter" && connecter()}
          />
          {erreur && <p className="text-[13px] text-rust">Renseignez identifiant et mot de passe.</p>}
          <button
            onClick={connecter}
            className="w-full bg-gold text-forest font-display font-bold rounded-xl py-3.5 mt-2 transition active:scale-[0.98]"
          >
            Se connecter
          </button>
        </div>
      </div>
    </div>
  );
}

/* Écran Admin — Tableau de bord                                      */
/* ------------------------------------------------------------------ */

function TableauBordAdmin({ reservations, onDeconnexion }) {
  const [onglet, setOnglet] = useState("vue");
  const [recherche, setRecherche] = useState("");

  const chiffreAffaires = reservations.reduce((s, r) => s + r.montant, 0);
  const totalPassagers = reservations.reduce((s, r) => s + r.passagers, 0);

  const trajets = useMemo(() => {
    const map = {};
    reservations.forEach((r) => {
      const cle = `${r.depart} → ${r.destination}`;
      map[cle] = (map[cle] || 0) + 1;
    });
    return Object.entries(map).sort((a, b) => b[1] - a[1]);
  }, [reservations]);

  const resultatsRecherche = reservations.filter((r) => {
    const q = recherche.trim().toLowerCase();
    if (!q) return false;
    return r.nom.toLowerCase().includes(q) || r.code.toLowerCase().includes(q);
  });

  const onglets = [
    { id: "vue", label: "Vue d'ensemble", icone: LayoutDashboard },
    { id: "trajets", label: "Trajets", icone: Bus },
    { id: "checkin", label: "Embarquement", icone: Search },
  ];

  return (
    <div className="min-h-[100dvh] bg-mist">
      <div className="bg-forest text-paper px-5 pt-6 pb-5">
        <div className="flex items-center justify-between max-w-md mx-auto">
          <div>
            <p className="text-[11px] uppercase tracking-wider text-paper/45 font-semibold">Tableau de bord</p>
            <h1 className="font-display font-bold text-[20px]">Easy Transport</h1>
          </div>
          <button
            onClick={onDeconnexion}
            className="text-[12.5px] text-paper/60 border border-paper/20 rounded-lg px-3 py-1.5 active:bg-paper/10 transition"
          >
            Déconnexion
          </button>
        </div>
      </div>

      <div className="max-w-md mx-auto px-5 -mt-3">
        <div className="flex bg-white rounded-xl border-2 border-forest/10 p-1">
          {onglets.map((o) => (
            <button
              key={o.id}
              onClick={() => setOnglet(o.id)}
              className={`flex-1 flex flex-col items-center gap-1 py-2 rounded-lg text-[11px] font-semibold transition ${
                onglet === o.id ? "bg-forest text-paper" : "text-forest/50"
              }`}
            >
              <o.icone size={15} />
              {o.label}
            </button>
          ))}
        </div>
      </div>

      <div className="max-w-md mx-auto px-5 py-5">
        {onglet === "vue" && (
          <div className="space-y-4">
            <div className="grid grid-cols-2 gap-3">
              <div className="bg-white rounded-2xl p-4 border-2 border-forest/10">
                <TrendingUp size={16} className="text-gold mb-2" />
                <p className="font-display font-bold text-[20px] text-ink leading-none">{formatFCFA(chiffreAffaires)}</p>
                <p className="text-[11.5px] text-ink/45 mt-1">Chiffre d'affaires</p>
              </div>
              <div className="bg-white rounded-2xl p-4 border-2 border-forest/10">
                <Ticket size={16} className="text-forest mb-2" />
                <p className="font-display font-bold text-[20px] text-ink leading-none">{reservations.length}</p>
                <p className="text-[11.5px] text-ink/45 mt-1">Réservations</p>
              </div>
            </div>
            <div className="bg-white rounded-2xl p-4 border-2 border-forest/10">
              <div className="flex items-center gap-2 mb-1">
                <Users size={15} className="text-forest" />
                <p className="text-[13px] font-semibold text-ink">{totalPassagers} passagers transportés</p>
              </div>
              <p className="text-[12px] text-ink/45">Sur {trajets.length} trajet(s) distinct(s)</p>
            </div>

            <div>
              <p className="text-[11px] uppercase tracking-[0.14em] text-forest/60 font-semibold mb-2.5">
                Dernières réservations
              </p>
              <div className="space-y-2">
                {reservations.length === 0 && (
                  <p className="text-[13.5px] text-ink/40 bg-white rounded-xl p-4 border-2 border-forest/10">
                    Aucune réservation pour le moment.
                  </p>
                )}
                {[...reservations].reverse().slice(0, 6).map((r) => (
                  <div key={r.code} className="bg-white rounded-xl p-3.5 border-2 border-forest/10 flex items-center justify-between">
                    <div className="min-w-0">
                      <p className="font-semibold text-[13.5px] text-ink truncate">{r.nom}</p>
                      <p className="text-[12px] text-ink/45 truncate">{r.depart} → {r.destination}</p>
                    </div>
                    <span className="font-mono text-[11.5px] font-bold text-forest shrink-0 ml-2">{r.code}</span>
                  </div>
                ))}
              </div>
            </div>
          </div>
        )}

        {onglet === "trajets" && (
          <div className="space-y-2.5">
            <p className="text-[11px] uppercase tracking-[0.14em] text-forest/60 font-semibold mb-1">
              Trajets & départs
            </p>
            {trajets.length === 0 && (
              <p className="text-[13.5px] text-ink/40 bg-white rounded-xl p-4 border-2 border-forest/10">
                Aucun trajet enregistré.
              </p>
            )}
            {trajets.map(([nom, count]) => (
              <div key={nom} className="bg-white rounded-xl p-4 border-2 border-forest/10 flex items-center justify-between">
                <span className="flex items-center gap-2.5 font-semibold text-[14px] text-ink">
                  <Bus size={15} className="text-forest shrink-0" />
                  {nom}
                </span>
                <span className="text-[12.5px] text-paper bg-forest rounded-full px-2.5 py-1 font-semibold">
                  {count} rés.
                </span>
              </div>
            ))}
          </div>
        )}

        {onglet === "checkin" && (
          <div className="space-y-4">
            <div className="relative">
              <Search size={16} className="absolute left-3.5 top-1/2 -translate-y-1/2 text-forest/40" />
              <input
                className="w-full rounded-xl border-2 border-forest/15 bg-white pl-10 pr-4 py-3 text-[14.5px] outline-none focus:border-gold transition"
                placeholder="Nom du passager ou code (ET-XXXXXX)"
                value={recherche}
                onChange={(e) => setRecherche(e.target.value)}
              />
            </div>

            {recherche.trim() && resultatsRecherche.length === 0 && (
              <p className="text-[13.5px] text-ink/40 bg-white rounded-xl p-4 border-2 border-forest/10">
                Aucune réservation correspondante.
              </p>
            )}

            <div className="space-y-2.5">
              {resultatsRecherche.map((r) => (
                <div key={r.code} className="bg-white rounded-2xl p-4 border-2 border-forest/10">
                  <div className="flex items-center justify-between mb-2.5">
                    <p className="font-display font-bold text-[15px] text-ink">{r.nom}</p>
                    <span className="font-mono text-[12px] font-bold text-forest">{r.code}</span>
                  </div>
                  <p className="text-[12.5px] text-ink/50 mb-3">
                    {r.depart} → {r.destination} · {r.date} · {r.passagers} passager(s)
                  </p>
                  <button className="w-full bg-forest text-paper text-[13.5px] font-semibold rounded-lg py-2.5 flex items-center justify-center gap-1.5 active:scale-[0.98] transition">
                    <CheckCircle2 size={15} />
                    Valider l'embarquement
                  </button>
                </div>
              ))}
            </div>
          </div>
        )}
      </div>
    </div>
  );
}

        @import url('https://fonts.googleapis.com/css2?family=Space+Grotesk:wght@500;600;700&family=Inter:wght@400;500;600;700&family=IBM+Plex+Mono:wght@600;700&display=swap');

        .text-forest { color: #0E3B2E; }
        .bg-forest { background-color: #0E3B2E; }
        .border-forest { border-color: #0E3B2E; }
        .text-gold { color: #E8A33D; }

        .ring-gold\\/15 { --tw-ring-color: rgba(232,163,61,0.15); }
      `}</style>

      {ecran === "accueil" && (
        <EcranAccueil
          onChoix={(role) => setEcran(role === "client" ? "formulaire" : "admin-login")}
        />
      )}

      {ecran === "formulaire" && (
        <FormulaireReservation
          onRetour={retourAccueil}
          onSuivant={(donnees) => {
            setEnCours(donnees);
            setEcran("resultats");
          }}
        />
      )}

      {ecran === "resultats" && enCours && (
        <EcranResultats
          reservation={enCours}
          onRetour={() => setEcran("formulaire")}
          onChoisir={(donnees) => {
            setEnCours(donnees);
            setEcran("paiement");
          }}
        />
      )}

      {ecran === "paiement" && enCours && enCours.trajet && (
        <EcranPaiement

          onRetour={() => setEcran("formulaire")}
          onPaye={(montant) => {

        <BilletNumerique billet={dernierBillet} onAccueil={retourAccueil} />
      )}

      {ecran === "admin-login" && (
        <ConnexionAdmin onRetour={retourAccueil} onConnecte={() => setEcran("admin-dashboard")} />
      )}

      {ecran === "admin-dashboard" && (
        <TableauBordAdmin reservations={reservations} onDeconnexion={retourAccueil} />
      )}
    </div>
  );
}
